cmake_minimum_required(VERSION 3.20)
project(KartDriftOffline VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(drift_core STATIC core/Game.cpp core/Scene.cpp core/Audio.cpp)
target_include_directories(drift_core PUBLIC core)
if(APPLE)
  enable_language(OBJCXX)
  add_executable(KartDriftOffline MACOSX_BUNDLE platform/apple/App.mm)
  target_compile_options(KartDriftOffline PRIVATE -fobjc-arc)
  target_link_libraries(KartDriftOffline PRIVATE drift_core "-framework Metal" "-framework MetalKit" "-framework QuartzCore" "-framework Foundation" "-framework AudioToolbox")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_link_libraries(KartDriftOffline PRIVATE "-framework UIKit")
    set_target_properties(KartDriftOffline PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/platform/apple/Info-iOS.plist")
  else()
    target_link_libraries(KartDriftOffline PRIVATE "-framework AppKit")
    set_target_properties(KartDriftOffline PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/platform/apple/Info-Mac.plist")
  endif()
  set_target_properties(KartDriftOffline PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.kartdrift.offline.native MACOSX_BUNDLE_BUNDLE_NAME "KartDrift Offline" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.0" MACOSX_BUNDLE_BUNDLE_VERSION 1)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT ANDROID)
  enable_testing()
  add_executable(race_test tests/race_test.cpp)
  target_compile_options(race_test PRIVATE -UNDEBUG)
  target_link_libraries(race_test PRIVATE drift_core)
  add_test(NAME offline_races COMMAND race_test)
endif()
